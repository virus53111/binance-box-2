// NEXUS ALPHA RAVE-type MEXC radar: obscure non-major-CEX markets with a compressed base and early ignition.
// "RAVE-type" describes market structure only. It is not a forecast that a token will repeat RAVE's move.
const RAVE_MEXC_MAJOR_FALLBACK=new Set(['BTC','ETH','BNB','SOL','XRP','DOGE','ADA','TRX','LINK','TAO','HBAR','CRV','DOT','LTC','BCH','AVAX','SUI','TON','SHIB','UNI','ATOM','NEAR','ETC','APT','ARB','OP','FIL','ICP','XLM','AAVE','MKR','PEPE']);
const RAVE_MEXC_NON_TOKEN_RE=/(?:^USD[TC]?|^USDE$|^FDUSD$|^EUR|^GBP|^JPY|XAUT|PAXG|GOLD|SILVER|XAG|XAU|STOCK|INDEX|\(|\))/i;
function raveMajorPresence(base){const s=normalizeSymbol(base),out=[];for(const[exchange,set]of Object.entries(cexSymbolSets||{}))if(set?.has(s))out.push(exchange);if(!out.length&&RAVE_MEXC_MAJOR_FALLBACK.has(s))out.push('major-cex-fallback');return uniq(out)}
function raveDailyStats(rows,current){const k=(Array.isArray(rows)?rows:[]).map(x=>({at:marketNum(x?.[0]),o:marketNum(x?.[1]),h:marketNum(x?.[2]),l:marketNum(x?.[3]),c:marketNum(x?.[4]),v:marketNum(x?.[7])||marketNum(x?.[5])*marketNum(x?.[4])})).filter(x=>x.c>0&&x.h>0&&x.l>0);if(k.length<3)return null;const last=current||k.at(-1).c,windowStats=n=>{const a=k.slice(-Math.min(n,k.length)),hi=Math.max(...a.map(x=>x.h)),lo=Math.min(...a.map(x=>x.l));return{hi,lo,rangePct:lo>0?(hi/lo-1)*100:999,pos:hi>lo?(last-lo)/(hi-lo):.5,floorPct:lo>0?(last/lo-1)*100:999}};const d7=windowStats(7),d30=windowStats(30),d90=windowStats(90),hiAll=Math.max(...k.map(x=>x.h)),loAll=Math.min(...k.map(x=>x.l)),ageDays=Math.max(1,Math.round((k.at(-1).at-k[0].at)/86400000)+1),athMultiple=last>0?hiAll/last:0,lifetimeMultiple=loAll>0?hiAll/loAll:0;return{bars:k.length,ageDays,firstAt:k[0].at,lastAt:k.at(-1).at,range7Pct:+d7.rangePct.toFixed(1),range30Pct:+d30.rangePct.toFixed(1),range90Pct:+d90.rangePct.toFixed(1),floorDistance90Pct:+d90.floorPct.toFixed(1),rangePosition90:+d90.pos.toFixed(3),athMultiple:+athMultiple.toFixed(2),lifetimeMultiple:+lifetimeMultiple.toFixed(2),priorMoonshot:athMultiple>=8||lifetimeMultiple>=25,nearFloor:d90.pos<=.35||d90.floorPct<=35,baseCompressed:d7.rangePct<=35&&d30.rangePct<=80}}
computeMexcExtreme=function computeMexcRaveExtreme(m){
  const d=m.deep||{},qv=marketNum(m.quoteVolume24h),spread=marketNum(d.spreadPct),depth2=marketNum(d.depth2Usd),depth5=marketNum(d.depth5Usd),imb=marketNum(d.bidAskImbalance2),k=d.kline||{},daily=d.daily||null,major=raveMajorPresence(m.base),flags=[],reasons=[];
  const nonToken=RAVE_MEXC_NON_TOKEN_RE.test(str(m.base)),wrongQuote=str(m.quote).toUpperCase()!=='USDT';let auth=0;
  if(qv>=30000)auth+=15;else flags.push('low_24h_volume');if(qv>=100000&&qv<=15000000)auth+=10;if(spread>0&&spread<=1.2)auth+=20;else if(spread<=2.5)auth+=8;else flags.push('wide_spread');if(depth2>=2500)auth+=15;else flags.push('thin_2pct_book');if(depth2>=8000&&depth2<=250000)auth+=10;if(k.bars>=24)auth+=15;if((k.wickRisk||0)<=1)auth+=15;else flags.push('wick_anomaly');auth=Math.round(clamp(auth,0,100));
  let score=0;
  if(!major.length){score+=20;reasons.push('mexc_only_no_major_cex')}else flags.push('major_cex_asset');
  if(daily){if(daily.ageDays>=7&&daily.ageDays<=540){score+=10;reasons.push('young_or_mid_age_market')}else if(daily.ageDays<=900)score+=4;if(daily.nearFloor){score+=15;reasons.push('near_90d_floor')}if(daily.range7Pct<=25){score+=10;reasons.push('tight_7d_base')}else if(daily.range7Pct<=40)score+=5;if(daily.range30Pct<=55){score+=9;reasons.push('compressed_30d_base')}else if(daily.range30Pct<=85)score+=4;if(daily.priorMoonshot){score-=30;flags.push('historic_extreme_move_already_happened')}}else flags.push('daily_history_pending');
  if(k.volAccel>=1.8){score+=10;reasons.push('volume_ignition')}if(k.volAccel>=3)score+=6;if(Math.abs(k.price3h||0)<=12&&k.volAccel>=1.8){score+=8;reasons.push('volume_before_price')}if(k.atrPct>0&&k.atrPct<=4.5){score+=7;reasons.push('hourly_compression')}if(imb>=1.15&&imb<=3.8){score+=7;reasons.push('bid_depth_advantage')}if(depth2>=3000&&depth2<=120000){score+=7;reasons.push('explosive_but_real_book')}else if(depth2>120000&&depth2<=300000)score+=3;
  const energy=depth2>0?qv/depth2:0;if(energy>=2&&energy<=80){score+=8;reasons.push('high_turnover_vs_depth')}else if(energy>80&&energy<=150)score+=2;else if(energy>150){score-=10;flags.push('turnover_depth_anomaly')}
  if(auth>=70)score+=7;else if(auth<50)score-=20;
  const alreadyMoved=Math.abs(k.price3h||0)>30||Math.abs(k.price24h||0)>70||Math.abs(m.change24h||0)>70;if(alreadyMoved){score-=35;flags.push('already_expanded')}
  if(qv>25000000){score-=20;flags.push('too_large_for_extreme_profile')}if(nonToken)flags.push('non_token_or_synthetic');if(wrongQuote)flags.push('non_usdt_pair');
  const eligible=!major.length&&!nonToken&&!wrongQuote&&!alreadyMoved&&qv>=25000&&qv<=25000000&&auth>=45&&depth2>=2000&&!(daily?.priorMoonshot);
  if(!eligible)score=Math.min(score,49);score=Math.round(clamp(score,0,100));
  let stage='NOT_READY';if(eligible&&daily?.baseCompressed&&daily?.nearFloor&&k.volAccel>=1.8&&Math.abs(k.price3h||0)<=12)stage='IGNITION';else if(eligible&&daily?.baseCompressed&&daily?.nearFloor)stage='DORMANT_BASE';else if(eligible&&k.volAccel>=2&&Math.abs(k.price3h||0)<=18)stage='EARLY_FLOW';
  const raveLike=eligible&&score>=70&&(stage==='IGNITION'||stage==='DORMANT_BASE'||stage==='EARLY_FLOW');
  return{score,raveDna:score,stage,raveLike,eligible,authenticity:auth,riskGate:eligible?(auth>=70?'pass':'caution'):'fail',reasons:uniq(reasons),flags:uniq(flags),alreadyMoved,majorCexCount:major.length,majorCex:major,profile:{ageDays:daily?.ageDays||0,range7Pct:daily?.range7Pct??null,range30Pct:daily?.range30Pct??null,floorDistance90Pct:daily?.floorDistance90Pct??null,rangePosition90:daily?.rangePosition90??null,athMultiple:daily?.athMultiple??null,priorMoonshot:Boolean(daily?.priorMoonshot),energyRatio:+energy.toFixed(1),quoteVolume24h:qv,depth2Usd:depth2},method:'RAVE-type structure score: no major-CEX presence, compressed base near a multi-week floor, early volume ignition before price expansion, and a real but relatively elastic order book. It is not a return prediction.'};
};
