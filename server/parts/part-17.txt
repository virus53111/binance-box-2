sourceState.marketmap=sourceState.marketmap||{label:'Global Market Map',ok:false,lastOk:0,lastError:'',latency:null};
const marketMapCache=new Map();let marketGlobalCache={at:0,data:null};
const marketShortHistory=new Map();let paprikaCache={at:0,rows:[]};let marketWarmRunning=false;const marketWarmStartedAt=now();
const marketMonthlyCache=new Map();const marketMonthlyQueue=[];const marketMonthlyQueued=new Set();let marketMonthlyRunning=false;let marketPersistenceReady=false;
const strictMarketNumber=v=>v===null||v===undefined||v===''||!Number.isFinite(Number(v))?null:Number(v);
function normalizeKey(v){return str(v).toLowerCase().replace(/[^a-z0-9]+/g,'')}
function recordMarketSnapshots(coins,at=now()){const cut=at-25*60_000;for(const c of coins){if(!c?.id||!c?.price)continue;const arr=marketShortHistory.get(c.id)||[];const last=arr[arr.length-1];if(!last||Math.abs(last.at-at)>20_000)arr.push({at,price:c.price});while(arr.length&&arr[0].at<cut)arr.shift();marketShortHistory.set(c.id,arr)}}
function shortChange(id,current,minutes){const arr=marketShortHistory.get(id)||[];if(!current||arr.length<2)return null;const target=now()-minutes*60_000;let best=null;for(const p of arr){if(p.at<=target)best=p;else break}if(!best||!best.price)return null;return(current/best.price-1)*100}
async function ensureMarketPersistence(){if(marketPersistenceReady||!pool)return Boolean(pool);await pool.query(`CREATE TABLE IF NOT EXISTS market_price_frames (ts bigint PRIMARY KEY, prices jsonb NOT NULL)`);await pool.query(`CREATE TABLE IF NOT EXISTS market_monthly_cache (coin_id text PRIMARY KEY, change_30d double precision, updated_at bigint NOT NULL)`);marketPersistenceReady=true;return true}
async function restoreMarketPersistence(){if(!await ensureMarketPersistence())return;try{const cut=now()-25*60_000;const {rows}=await pool.query('SELECT ts,prices FROM market_price_frames WHERE ts >= $1 ORDER BY ts ASC',[cut]);for(const row of rows){for(const [id,price] of Object.entries(row.prices||{})){const p=strictMarketNumber(price);if(p===null||p<=0)continue;const arr=marketShortHistory.get(id)||[];arr.push({at:Number(row.ts),price:p});marketShortHistory.set(id,arr)}}const monthly=await pool.query('SELECT coin_id,change_30d,updated_at FROM market_monthly_cache WHERE updated_at >= $1',[now()-36*60*60_000]);for(const row of monthly.rows)marketMonthlyCache.set(String(row.coin_id),{value:strictMarketNumber(row.change_30d),at:Number(row.updated_at)});}catch(error){console.error('[market persistence restore]',compactError(error))}}
async function persistMarketFrame(coins){if(!coins?.length||!await ensureMarketPersistence())return;const ts=now(),prices={};for(const c of coins)if(c?.id&&c?.price>0)prices[c.id]=c.price;try{await pool.query('INSERT INTO market_price_frames (ts,prices) VALUES ($1,$2) ON CONFLICT (ts) DO NOTHING',[ts,prices]);await pool.query('DELETE FROM market_price_frames WHERE ts < $1',[ts-30*60_000]);}catch(error){console.error('[market frame persist]',compactError(error))}}
async function persistMonthly(id,value,at){if(!await ensureMarketPersistence())return;try{await pool.query('INSERT INTO market_monthly_cache (coin_id,change_30d,updated_at) VALUES ($1,$2,$3) ON CONFLICT (coin_id) DO UPDATE SET change_30d=EXCLUDED.change_30d,updated_at=EXCLUDED.updated_at',[id,value,at]);}catch{}}
async function getPaprikaRows(){if(paprikaCache.rows.length&&now()-paprikaCache.at<5*60_000)return paprikaCache.rows;try{const raw=await fetchJson('https://api.coinpaprika.com/v1/tickers?quotes=USD',{timeout:15000});const rows=(Array.isArray(raw)?raw:[]).map(x=>({rank:num(x?.rank),symbol:str(x?.symbol).toUpperCase(),nameKey:normalizeKey(x?.name),q:x?.quotes?.USD||{}}));paprikaCache={at:now(),rows};return rows}catch{return paprikaCache.rows||[]}}
function paprikaFor(c,rows){const sym=str(c?.symbol).toUpperCase(),nameKey=normalizeKey(c?.name);const same=rows.filter(x=>x.symbol===sym);if(!same.length)return null;return same.sort((a,b)=>{const an=a.nameKey===nameKey?0:1,bn=b.nameKey===nameKey?0:1;if(an!==bn)return an-bn;return Math.abs((a.rank||99999)-(c.rank||99999))-Math.abs((b.rank||99999)-(c.rank||99999))})[0]||null}
function normalizeMarketMapCoin(x){const nameid=str(x?.nameid);return{id:str(x?.id),rank:num(x?.rank),symbol:str(x?.symbol).toUpperCase(),name:str(x?.name),nameid,logo:nameid?`https://c2.coinlore.com/img/25x25/${encodeURIComponent(nameid)}.png`:'',price:num(x?.price_usd),marketCap:num(x?.market_cap_usd),volume24:num(x?.volume24),change1h:strictMarketNumber(x?.percent_change_1h),change24h:strictMarketNumber(x?.percent_change_24h),change7d:strictMarketNumber(x?.percent_change_7d),supply:num(x?.csupply),maxSupply:num(x?.msupply),priceBtc:num(x?.price_btc)}}
function monthlyFresh(entry){if(!entry)return false;const ttl=entry.value===null?60*60_000:24*60*60_000;return now()-entry.at<ttl}
function queueMonthly(coins){for(const c of coins||[]){if(!c?.id||!c?.price)continue;const entry=marketMonthlyCache.get(c.id);if(monthlyFresh(entry)||marketMonthlyQueued.has(c.id))continue;marketMonthlyQueued.add(c.id);marketMonthlyQueue.push({id:c.id,price:c.price})}runMonthlyWorker().catch(()=>{})}
async function fetchMonthlyChange(coin){const raw=await fetchJson(`https://api.coinlore.net/api/coin/ohlcv/?coin=${encodeURIComponent(coin.id)}`,{timeout:12000});const rows=Object.values(raw||{}).filter(v=>Array.isArray(v)&&v.length>=5&&Number(v[0])>0&&Number(v[4])>0).sort((a,b)=>Number(a[0])-Number(b[0]));if(rows.length<20)return null;const target=Math.floor((now()-30*24*60*60_000)/1000);let base=null;for(const row of rows){if(Number(row[0])<=target)base=row;else break}if(!base)base=rows[Math.max(0,rows.length-31)];const close=Number(base?.[4]);if(!close||!coin.price)return null;return(coin.price/close-1)*100}
async function runMonthlyWorker(){if(marketMonthlyRunning)return;marketMonthlyRunning=true;try{while(marketMonthlyQueue.length){const coin=marketMonthlyQueue.shift();marketMonthlyQueued.delete(coin.id);const at=now();let value=null;try{value=await fetchMonthlyChange(coin)}catch{}marketMonthlyCache.set(coin.id,{value,at});if(value!==null)persistMonthly(coin.id,value,at).catch(()=>{});marketMapCache.clear();await sleep(1050)}}finally{marketMonthlyRunning=false}}
function decorateMarketData(data){return{...data,coins:(data.coins||[]).map(c=>{const q=paprikaFor(c,paprikaCache.rows)?.q||{};const p15=strictMarketNumber(q.percent_change_15m);const monthly=marketMonthlyCache.get(c.id);return{...c,change5m:shortChange(c.id,c.price,5),change15m:p15!==null?p15:shortChange(c.id,c.price,15),change30d:monthlyFresh(monthly)?monthly.value:null}})}}
async function getMarketMapPage(start=0,limit=100){start=Math.max(0,Math.floor(start/100)*100);limit=clamp(num(limit)||100,1,100);const key=`${start}:${limit}`,cached=marketMapCache.get(key);if(cached&&now()-cached.at<30_000){queueMonthly(cached.data.coins);return decorateMarketData(cached.data)}const began=now();try{const[raw,paprika]=await Promise.all([fetchJson(`https://api.coinlore.net/api/tickers/?start=${start}&limit=${limit}`,{timeout:10000}),getPaprikaRows()]),base=(raw?.data||[]).map(normalizeMarketMapCoin).filter(x=>x.symbol&&x.price>0);recordMarketSnapshots(base);paprikaCache.rows=paprika;queueMonthly(base);const data={generatedAt:now(),start,limit,total:num(raw?.info?.coins_num),sourceTime:num(raw?.info?.time)*1000,coins:base};marketMapCache.set(key,{at:now(),data});touchSource('marketmap',true,began);return decorateMarketData(data)}catch(error){touchSource('marketmap',false,began,error);if(cached)return decorateMarketData(cached.data);throw error}}
async function getMarketGlobal(){if(marketGlobalCache.data&&now()-marketGlobalCache.at<60_000)return marketGlobalCache.data;try{const raw=await fetchJson('https://api.coinlore.net/api/global/',{timeout:9000}),data=Array.isArray(raw)?raw[0]||{}:raw||{};marketGlobalCache={at:now(),data};return data}catch{return marketGlobalCache.data||{}}}
