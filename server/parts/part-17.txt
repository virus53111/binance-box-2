// NEXUS ALPHA market-map feed: free market-cap rankings from CoinLore with short server cache.
sourceState.marketmap=sourceState.marketmap||{label:'Global Market Map',ok:false,lastOk:0,lastError:'',latency:null};
const marketMapCache=new Map();let marketGlobalCache={at:0,data:null};
function normalizeMarketMapCoin(x){const nameid=str(x?.nameid);return{id:str(x?.id),rank:num(x?.rank),symbol:str(x?.symbol).toUpperCase(),name:str(x?.name),nameid,logo:nameid?`https://c2.coinlore.com/img/25x25/${encodeURIComponent(nameid)}.png`:'',price:num(x?.price_usd),marketCap:num(x?.market_cap_usd),volume24:num(x?.volume24),change1h:num(x?.percent_change_1h),change24h:num(x?.percent_change_24h),change7d:num(x?.percent_change_7d),supply:num(x?.csupply),maxSupply:num(x?.msupply),priceBtc:num(x?.price_btc)}}
async function getMarketMapPage(start=0,limit=100){start=Math.max(0,Math.floor(start/100)*100);limit=clamp(num(limit)||100,1,100);const key=`${start}:${limit}`,cached=marketMapCache.get(key);if(cached&&now()-cached.at<45_000)return cached.data;const began=now();try{const raw=await fetchJson(`https://api.coinlore.net/api/tickers/?start=${start}&limit=${limit}`,{timeout:10000}),coins=(raw?.data||[]).map(normalizeMarketMapCoin).filter(x=>x.symbol&&x.price>0);const data={generatedAt:now(),start,limit,total:num(raw?.info?.coins_num),sourceTime:num(raw?.info?.time)*1000,coins};marketMapCache.set(key,{at:now(),data});touchSource('marketmap',true,began);return data}catch(error){touchSource('marketmap',false,began,error);if(cached)return cached.data;throw error}}
async function getMarketGlobal(){if(marketGlobalCache.data&&now()-marketGlobalCache.at<60_000)return marketGlobalCache.data;try{const raw=await fetchJson('https://api.coinlore.net/api/global/',{timeout:9000}),data=Array.isArray(raw)?raw[0]||{}:raw||{};marketGlobalCache={at:now(),data};return data}catch{return marketGlobalCache.data||{}}}
app.get('/api/market-map',async(req,res)=>{try{const start=clamp(num(req.query.start)||0,0,50_000),limit=clamp(num(req.query.limit)||100,1,100),[page,global]=await Promise.all([getMarketMapPage(start,limit),getMarketGlobal()]);res.json({...page,global,source:sourceState.marketmap,methodology:'Global crypto rankings are sorted by market capitalization by the upstream public market-data provider. Prices and ranks may differ slightly from individual exchanges.'})}catch(error){res.status(503).json({error:compactError(error),source:sourceState.marketmap})}});
