// NEXUS ALPHA market-map feed: free market-cap rankings + short snapshots + optional CoinPaprika enrichment.
sourceState.marketmap=sourceState.marketmap||{label:'Global Market Map',ok:false,lastOk:0,lastError:'',latency:null};
const marketMapCache=new Map();let marketGlobalCache={at:0,data:null};
const marketShortHistory=new Map();let paprikaCache={at:0,rows:[]};
function normalizeKey(v){return str(v).toLowerCase().replace(/[^a-z0-9]+/g,'')}
function recordMarketSnapshots(coins){const t=now(),cut=t-20*60_000;for(const c of coins){if(!c?.id||!c?.price)continue;const arr=marketShortHistory.get(c.id)||[];arr.push({at:t,price:c.price});while(arr.length&&arr[0].at<cut)arr.shift();marketShortHistory.set(c.id,arr)}}
function shortChange(id,current,minutes){const arr=marketShortHistory.get(id)||[];if(!current||arr.length<2)return null;const target=now()-minutes*60_000;let best=null;for(const p of arr){if(p.at<=target)best=p;else break}if(!best)return null;return (current/best.price-1)*100}
async function getPaprikaRows(){if(paprikaCache.rows.length&&now()-paprikaCache.at<5*60_000)return paprikaCache.rows;try{const raw=await fetchJson('https://api.coinpaprika.com/v1/tickers?quotes=USD',{timeout:15000});const rows=(Array.isArray(raw)?raw:[]).map(x=>({rank:num(x?.rank),symbol:str(x?.symbol).toUpperCase(),nameKey:normalizeKey(x?.name),q:x?.quotes?.USD||{}}));paprikaCache={at:now(),rows};return rows}catch{return paprikaCache.rows||[]}}
function paprikaFor(c,rows){const sym=str(c?.symbol).toUpperCase(),nameKey=normalizeKey(c?.name);const same=rows.filter(x=>x.symbol===sym);if(!same.length)return null;return same.sort((a,b)=>{const an=a.nameKey===nameKey?0:1,bn=b.nameKey===nameKey?0:1;if(an!==bn)return an-bn;return Math.abs((a.rank||99999)-(c.rank||99999))-Math.abs((b.rank||99999)-(c.rank||99999))})[0]||null}
function normalizeMarketMapCoin(x){const nameid=str(x?.nameid);return{id:str(x?.id),rank:num(x?.rank),symbol:str(x?.symbol).toUpperCase(),name:str(x?.name),nameid,logo:nameid?`https://c2.coinlore.com/img/25x25/${encodeURIComponent(nameid)}.png`:'',price:num(x?.price_usd),marketCap:num(x?.market_cap_usd),volume24:num(x?.volume24),change1h:num(x?.percent_change_1h),change24h:num(x?.percent_change_24h),change7d:num(x?.percent_change_7d),supply:num(x?.csupply),maxSupply:num(x?.msupply),priceBtc:num(x?.price_btc)}}
async function getMarketMapPage(start=0,limit=100){start=Math.max(0,Math.floor(start/100)*100);limit=clamp(num(limit)||100,1,100);const key=`${start}:${limit}`,cached=marketMapCache.get(key);if(cached&&now()-cached.at<30_000)return cached.data;const began=now();try{const [raw,paprika]=await Promise.all([fetchJson(`https://api.coinlore.net/api/tickers/?start=${start}&limit=${limit}`,{timeout:10000}),getPaprikaRows()]),base=(raw?.data||[]).map(normalizeMarketMapCoin).filter(x=>x.symbol&&x.price>0);recordMarketSnapshots(base);const coins=base.map(c=>{const p=paprikaFor(c,paprika),q=p?.q||{};return{...c,change5m:shortChange(c.id,c.price,5),change15m:Number.isFinite(Number(q.percent_change_15m))?Number(q.percent_change_15m):shortChange(c.id,c.price,15),change30d:Number.isFinite(Number(q.percent_change_30d))?Number(q.percent_change_30d):null}});const data={generatedAt:now(),start,limit,total:num(raw?.info?.coins_num),sourceTime:num(raw?.info?.time)*1000,coins};marketMapCache.set(key,{at:now(),data});touchSource('marketmap',true,began);return data}catch(error){touchSource('marketmap',false,began,error);if(cached)return cached.data;throw error}}
async function getMarketGlobal(){if(marketGlobalCache.data&&now()-marketGlobalCache.at<60_000)return marketGlobalCache.data;try{const raw=await fetchJson('https://api.coinlore.net/api/global/',{timeout:9000}),data=Array.isArray(raw)?raw[0]||{}:raw||{};marketGlobalCache={at:now(),data};return data}catch{return marketGlobalCache.data||{}}}
app.get('/api/market-map',async(req,res)=>{try{const start=clamp(num(req.query.start)||0,0,50_000),limit=clamp(num(req.query.limit)||100,1,100),[page,global]=await Promise.all([getMarketMapPage(start,limit),getMarketGlobal()]);res.json({...page,global,source:sourceState.marketmap,methodology:'Rankings/1h/24h/7d come from CoinLore. 5m is computed from NEXUS live snapshots; 15m/30d are enriched when the free supplemental source is available. Missing windows are returned as null, never fabricated.'})}catch(error){res.status(503).json({error:compactError(error),source:sourceState.marketmap})}});
