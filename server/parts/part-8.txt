const projectScanPersistBase=scanProjectToken;
scanProjectToken=async function scanAndPersistProjectToken(token){
  if(!projectSignalState.has(token.mint)&&Array.isArray(token.projectSignals)&&token.projectSignals.length){
    projectSignalState.set(token.mint,{fingerprints:new Set(token.projectSignals.map(projectFingerprint)),signals:token.projectSignals,initialized:true,lastScan:num(token.projectChannels?.lastScan),errors:token.projectChannels?.errors||[],okSources:num(token.projectChannels?.okSources),totalSources:num(token.projectChannels?.totalSources)});
  }
  await projectScanPersistBase(token);
  const st=projectSignalState.get(token.mint);
  if(st){token.projectSignals=st.signals;token.projectChannels={lastScan:st.lastScan,errors:st.errors,okSources:st.okSources,totalSources:st.totalSources};token.lastUpdatedAt=now()}
};
const projectIntelPersistBase=getCexIntel;
getCexIntel=function getCexIntelWithStoredProjectClaims(token){
  const current=projectIntelPersistBase(token);if(current.projectSignals?.length||!token.projectSignals?.length)return current;
  const claims=token.projectSignals,combined=[...claims,...(current.signals||[])].sort((a,b)=>(b.evidenceScore||0)-(a.evidenceScore||0)).slice(0,12),bestProject=claims[0]||null;
  return {...current,signals:combined,projectSignals:claims,projectChannels:token.projectChannels||current.projectChannels,projectClaimDetected:true,preCexCandidate:current.preCexCandidate||Boolean(!current.majorCexSymbolDetected&&bestProject?.evidenceScore>=74),evidenceScore:Math.max(current.evidenceScore||0,bestProject?.evidenceScore||0)};
};
