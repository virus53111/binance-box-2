pruneTokens=function pruneExpandedUniverse(){
  const cap=8000;if(tokens.size<=cap)return;
  const ordered=[...tokens.values()].sort((a,b)=>{
    const aClaim=(projectSignalState.get(a.mint)?.signals?.length||a.projectSignals?.length)?1:0,bClaim=(projectSignalState.get(b.mint)?.signals?.length||b.projectSignals?.length)?1:0;
    if(aClaim!==bClaim)return aClaim-bClaim;
    const aLinks=projectSourceList(a).length?1:0,bLinks=projectSourceList(b).length?1:0;if(aLinks!==bLinks)return aLinks-bLinks;
    const aL=num(a.dex?.liquidityUsd),bL=num(b.dex?.liquidityUsd);if((aL>=10000)!==(bL>=10000))return (aL>=10000?1:0)-(bL>=10000?1:0);
    return (a.lastUpdatedAt||0)-(b.lastUpdatedAt||0);
  });
  for(const token of ordered.slice(0,tokens.size-cap))tokens.delete(token.mint);
};

  const old=token.dex||{};token.dex={...old,dexId:old.dexId||str(row?.relationships?.dex?.data?.id)||'geckoterminal',pairAddress:old.pairAddress||str(a.address),url:old.url||safeUrl(`https://www.geckoterminal.com/solana/pools/${str(a.address)}`),priceUsd:num(old.priceUsd)||num(a.base_token_price_usd),marketCap:num(old.marketCap)||num(a.market_cap_usd),fdv:num(old.fdv)||num(a.fdv_usd),liquidityUsd:num(old.liquidityUsd)||num(a.reserve_in_usd),volumeM5:num(old.volumeM5)||num(a.volume_usd?.m5),volumeH1:num(old.volumeH1)||num(a.volume_usd?.h1),volumeH6:num(old.volumeH6)||num(a.volume_usd?.h6),txM5:num(old.txM5)||num(tx5.buys)+num(tx5.sells),buysM5:num(old.buysM5)||num(tx5.buys),sellsM5:num(old.sellsM5)||num(tx5.sells),txH1:num(old.txH1)||num(tx1.buys)+num(tx1.sells),buysH1:num(old.buysH1)||num(tx1.buys),sellsH1:num(old.sellsH1)||num(tx1.sells),changeM5:Number.isFinite(Number(old.changeM5))?num(old.changeM5):num(a.price_change_percentage?.m5),changeH1:Number.isFinite(Number(old.changeH1))?num(old.changeH1):num(a.price_change_percentage?.h1),boosts:num(old.boosts),pairCreatedAt:num(old.pairCreatedAt)||created};

async function enrichDormantDex(list){
  const chosen=list.filter(t=>ageMs(t)>=DORMANT_MIN_AGE_MS).sort((a,b)=>(num(b.dex?.volumeH1)-num(a.dex?.volumeH1))||(num(b.dex?.liquidityUsd)-num(a.dex?.liquidityUsd))).slice(0,90);
  for(const group of chunk(chosen.map(t=>t.mint),30)){
    if(!group.length)continue;
    try{const data=await fetchJson(`https://api.dexscreener.com/tokens/v1/solana/${group.join(',')}`,{timeout:9000}),pairs=Array.isArray(data)?data:[],best=new Map();for(const pair of pairs){const mint=str(pair?.baseToken?.address);if(!mint||!tokens.has(mint))continue;if(!best.has(mint)||num(pair?.liquidity?.usd)>num(best.get(mint)?.liquidity?.usd))best.set(mint,pair)}for(const[mint,pair]of best)applyDexPair(tokens.get(mint),pair)}catch(error){if(!compactError(error).includes('429'))sourceState.dormant.lastError=`DEX metadata: ${compactError(error)}`}
    await sleep(1000);
  }
}

async function pollDormantUniverse(){
  const started=now(),observed=[];let pagesOk=0,lastError='';
  try{
    for(const duration of ['24h','6h'])for(let page=1;page<=5;page++){
      try{const data=await fetchDormantPage(duration,page),inc=geckoTokenMap(data),rows=Array.isArray(data?.data)?data.data:[];if(!rows.length)break;pagesOk++;for(const row of rows){const token=upsertDormantPool(row,inc);if(token&&ageMs(token)>=DORMANT_MIN_AGE_MS&&ageMs(token)<=DORMANT_MAX_AGE_MS&&num(token.dex?.liquidityUsd)>=3000)observed.push(token)}}catch(error){lastError=compactError(error);if(page===1)break}
      await sleep(850);
    }
    if(observed.length)await enrichDormantDex(uniq(observed.map(t=>t.mint)).map(m=>tokens.get(m)).filter(Boolean));
    const persist=observed.filter(t=>projectSourceList(t).length||num(t.dex?.liquidityUsd)>=10000).sort((a,b)=>(num(b.dex?.volumeH1)-num(a.dex?.volumeH1))||(b.score-a.score)).slice(0,100);
    if(pool)for(const token of persist){try{await persistToken(token)}catch{}}
    if(pagesOk){touchSource('dormant',true,started);sourceState.dormant.lastError=''}else touchSource('dormant',false,started,lastError||'No legacy pool pages');
    pruneTokens();
  }catch(error){touchSource('dormant',false,started,error)}
}
