// NEXUS ALPHA outcome lab: records setup-time features and later realized path for calibration.
const extremeDexAlerted=new Set();
async function initExtremeLab(){if(!pool)return;try{await pool.query(`CREATE TABLE IF NOT EXISTS extreme_observations (id text PRIMARY KEY, market text NOT NULL, asset_key text NOT NULL, observed_at bigint NOT NULL, entry_price double precision NOT NULL, setup_score integer NOT NULL, risk_gate text NOT NULL, features jsonb NOT NULL, max_price double precision NOT NULL, min_price double precision NOT NULL, last_price double precision NOT NULL, last_seen bigint NOT NULL, outcome jsonb NOT NULL DEFAULT '{}'::jsonb)`);await pool.query('CREATE INDEX IF NOT EXISTS extreme_observations_time_idx ON extreme_observations(observed_at DESC)')}catch(error){console.error('Extreme lab init:',compactError(error))}}
function obsId(market,key){return`${market}:${key}:${Math.floor(now()/(6*3600_000))}`}
async function saveExtremeObservation(market,key,price,score,gate,features){if(!pool||!price||score<55||gate==='fail')return;const id=obsId(market,key),ts=now(),safeFeatures=JSON.parse(JSON.stringify(features||{}));try{await pool.query(`INSERT INTO extreme_observations(id,market,asset_key,observed_at,entry_price,setup_score,risk_gate,features,max_price,min_price,last_price,last_seen,outcome) VALUES($1,$2,$3,$4,$5,$6,$7,$8,$5,$5,$5,$4,'{}'::jsonb) ON CONFLICT(id) DO UPDATE SET setup_score=GREATEST(extreme_observations.setup_score,EXCLUDED.setup_score),risk_gate=EXCLUDED.risk_gate,features=EXCLUDED.features,last_seen=EXCLUDED.last_seen`,[id,market,key,ts,price,Math.round(score),gate,safeFeatures])}catch{}}
async function captureExtremeSetups(){
  for(const token of [...tokens.values()]){if(!token.dex?.priceUsd)continue;const e=computeDexExtreme(token);if(e.score>=55&&e.riskGate!=='fail')await saveExtremeObservation('solana',token.mint,marketNum(token.dex.priceUsd),e.score,e.riskGate,{setup:e.setup,reasons:e.reasons,risk:token.risk,riskLevel:token.riskLevel,liquidityUsd:token.dex.liquidityUsd,volumeH1:token.dex.volumeH1,ageMs:ageMs(token),cex:{major:e.cex.majorCexSymbolDetected,projectClaim:e.cex.projectClaimDetected}});if(e.score>=82&&e.riskGate==='pass'&&!extremeDexAlerted.has(token.mint)){extremeDexAlerted.add(token.mint);addAlert({severity:'critical',type:'extreme',mint:token.mint,symbol:token.symbol,title:'EXTREME_ASYMMETRY',message:`Extreme setup ${e.score}; risk gate PASS`,score:e.score})}}
  for(const m of mexcMarkets.values()){const e=m.extreme;if(!e||!m.price)continue;if(e.score>=55&&e.riskGate!=='fail')await saveExtremeObservation('mexc',m.symbol,m.price,e.score,e.riskGate,{extreme:e,change24h:m.change24h,quoteVolume24h:m.quoteVolume24h,deep:m.deep})}
}
async function updateExtremeOutcomes(){if(!pool)return;try{const{rows}=await pool.query('SELECT id,market,asset_key,observed_at,entry_price,max_price,min_price FROM extreme_observations WHERE observed_at>$1 ORDER BY observed_at DESC LIMIT 800',[now()-48*3600_000]);for(const r of rows){let price=0;if(r.market==='solana')price=marketNum(tokens.get(r.asset_key)?.dex?.priceUsd);else if(r.market==='mexc')price=marketNum(mexcMarkets.get(r.asset_key)?.price);if(!price)continue;const maxPrice=Math.max(marketNum(r.max_price),price),minPrice=Math.min(marketNum(r.min_price)||price,price),entry=marketNum(r.entry_price),age=now()-Number(r.observed_at),outcome={ageMs:age,returnPct:entry>0?(price/entry-1)*100:0,maxReturnPct:entry>0?(maxPrice/entry-1)*100:0,maxDrawdownPct:entry>0?(minPrice/entry-1)*100:0,completed24h:age>=24*3600_000};await pool.query('UPDATE extreme_observations SET max_price=$2,min_price=$3,last_price=$4,last_seen=$5,outcome=$6 WHERE id=$1',[r.id,maxPrice,minPrice,price,now(),outcome])}await pool.query('DELETE FROM extreme_observations WHERE observed_at<$1',[now()-180*24*3600_000])}catch{}}
app.get('/api/extreme/calibration',async(_req,res)=>{if(!pool)return res.json({available:false,count:0});try{const{rows}=await pool.query(`SELECT market,COUNT(*)::int AS n,AVG(setup_score)::float AS avg_score,AVG(CASE WHEN (outcome->>'completed24h')::boolean THEN (outcome->>'maxReturnPct')::float END)::float AS avg_max_24h,SUM(CASE WHEN (outcome->>'completed24h')::boolean AND (outcome->>'maxReturnPct')::float>=20 THEN 1 ELSE 0 END)::int AS hit20,SUM(CASE WHEN (outcome->>'completed24h')::boolean AND (outcome->>'maxReturnPct')::float>=100 THEN 1 ELSE 0 END)::int AS hit100 FROM extreme_observations GROUP BY market`);res.json({available:true,generatedAt:now(),markets:rows,method:'Observed setup outcomes; descriptive calibration only, not a forecast.'})}catch(error){res.status(503).json({available:false,error:compactError(error)})}});
setTimeout(()=>initExtremeLab().catch(()=>{}),12_000);setTimeout(()=>captureExtremeSetups().catch(()=>{}),90_000);setInterval(()=>captureExtremeSetups().catch(()=>{}),10*60_000);setInterval(()=>updateExtremeOutcomes().catch(()=>{}),5*60_000);
