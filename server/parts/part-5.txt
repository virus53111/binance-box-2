function logRadarHealth(label) {
  console.log(label, JSON.stringify({
    storage: pool ? 'postgres' : 'memory',
    stats: statsSnapshot(),
    sources: Object.fromEntries(Object.entries(sourceState).map(([key, value]) => [key, { ok: value.ok, lastOk: value.lastOk, lastError: value.lastError, latency: value.latency }]))
  }));
}

let lastRpcRequestAt = 0;
rpc = async function rpcWithFallback(method, params) {
  const started = now();
  const urls = uniq([
    'https://solana-rpc.publicnode.com',
    SOLANA_RPC_URL,
    'https://api.mainnet.solana.com'
  ]);
  let lastError = '';
  for (const url of urls) {
    try {
      const gap = 300 - (now() - lastRpcRequestAt);
      if (gap > 0) await sleep(gap);
      lastRpcRequestAt = now();
      const data = await fetchJson(url, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ jsonrpc: '2.0', id: 1, method, params }),
        timeout: 9000
      });
      if (data?.error) throw new Error(data.error.message || 'RPC error');
      touchSource('solana', true, started);
      return data?.result;
    } catch (error) {
      lastError = compactError(error);
      if (lastError.includes('429')) await sleep(700);
    }
  }
  touchSource('solana', false, started, lastError);
  throw new Error(lastError || 'All Solana RPC endpoints failed');
};

let dexBackoffUntil = 0;
pollDexPairs = async function pollDexPairsThrottled() {
  const started = now();
  if (now() < dexBackoffUntil) return;
  try {
    const candidates = [...tokens.values()]
      .filter(t => ageMs(t) < 24 * 3600_000)
      .sort((a, b) => (b.score - a.score) || (b.discoveredAt - a.discoveredAt))
      .slice(0, 90);
    const groups = chunk(candidates.map(t => t.mint), 30).slice(0, 3);
    for (const group of groups) {
      if (!group.length) continue;
      let data;
      try {
        data = await fetchJson(`https://api.dexscreener.com/tokens/v1/solana/${group.join(',')}`, { timeout: 9000 });
      } catch (error) {
        const message = compactError(error);
        if (message.includes('429')) {
          dexBackoffUntil = now() + 60_000;
          touchSource('dex', false, started, message);
          return;
        }
        throw error;
      }
      const pairs = Array.isArray(data) ? data : [];
      const best = new Map();
      for (const pair of pairs) {
        const mint = str(pair?.baseToken?.address);
        if (!mint || !tokens.has(mint)) continue;
        const liq = num(pair?.liquidity?.usd);
        if (!best.has(mint) || liq > num(best.get(mint)?.liquidity?.usd)) best.set(mint, pair);
      }
      for (const [mint, pair] of best) applyDexPair(tokens.get(mint), pair);
      await sleep(850);
    }
    touchSource('dex', true, started);
  } catch (error) {
    touchSource('dex', false, started, error);
  }
};

async function checkGoPlusHealth() {
  const started = now();
  try {
    const headers = GOPLUS_TOKEN ? { authorization: `Bearer ${GOPLUS_TOKEN}` } : {};
    const data = await fetchJson('https://api.gopluslabs.io/api/v1/solana/token_security?contract_addresses=So11111111111111111111111111111111111111112', { headers, timeout: 8000 });
    if (!data || typeof data !== 'object') throw new Error('Invalid GoPlus response');
    touchSource('goplus', true, started);
  } catch (error) {
    touchSource('goplus', false, started, error);
  }
}

pollBinance = async function pollBinanceCurrent() {
  const started = now();
  const urls = [
    'https://www.binance.com/bapi/composite/v1/public/cms/article/list/query?type=1&pageNo=1&pageSize=50',
    'https://www.binance.com/bapi/composite/v1/public/cms/article/list/query?type=1&category=48&pageNo=1&pageSize=50',
    'https://www.binance.com/bapi/apex/v1/public/apex/cms/article/list/query?type=1&pageNo=1&pageSize=50'
  ];
  let lastError = '';
  for (const url of urls) {
    try {
      const data = await fetchJson(url, { timeout: 8000, headers: { clienttype: 'web', referer: 'https://www.binance.com/en/support/announcement/list/48' } });
      const rows = parseBinanceRows(data);
      if (!rows.length) throw new Error('No announcement rows');
      const normalized = rows.slice(0, 60).map(x => ({
        id: str(x?.id || x?.code || x?.articleId),
        title: str(x?.title),
        releaseDate: normalizeTs(x?.releaseDate || x?.publishDate || x?.date),
        code: str(x?.code || x?.id)
      })).filter(x => x.title);
      if (!normalized.length) throw new Error('No announcement titles');
      binanceArticles.splice(0, binanceArticles.length, ...normalized);
      matchBinanceMentions();
      touchSource('binance', true, started);
      return;
    } catch (error) {
      lastError = compactError(error);
    }
  }
  touchSource('binance', false, started, lastError);
};

async function boot() {
  try { await initDb(); } catch (error) { console.error('DB init:', compactError(error)); }
  await Promise.allSettled([
    pollPump(),
    pollDexProfiles(),
    pollBinance(),
    rpc('getSlot', [{ commitment: 'confirmed' }]),
    checkGoPlusHealth()
  ]);
  await pollDexPairs().catch(() => {});
  logRadarHealth('Murdili Radar startup snapshot');

  setInterval(() => pollPump().catch(() => {}), 7000);
  setInterval(() => pollDexProfiles().catch(() => {}), 60_000);
  setInterval(() => pollDexPairs().catch(() => {}), 30_000);
  setInterval(() => pollBinance().catch(() => {}), 20_000);
  enrichmentLoop().catch(() => {});
  persistenceLoop().catch(() => {});
  setTimeout(() => logRadarHealth('Murdili Radar health snapshot'), 60_000);

  app.listen(PORT, '0.0.0.0', () => console.log(`Murdili Radar API listening on ${PORT}`));
}

boot().catch(error => {
  console.error(error);
  process.exit(1);
});
