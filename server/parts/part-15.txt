// Vetted DEX refresh tuning: declare discovery healthy after the fast trust pass; deep checks continue separately.
refreshVettedUniverse=async function refreshVettedUniverseFast(){
  const started=now();
  try{
    const base=[...tokens.values()].filter(t=>t.dex?.pairAddress&&ageMs(t)>=VETTED_MIN_AGE_MS&&ageMs(t)<=VETTED_MAX_AGE_MS&&!getCexIntel(t).majorCexSymbolDetected&&marketNum(t.dex?.liquidityUsd)>=15000).sort((a,b)=>(marketNum(b.dex?.volumeH1)-marketNum(a.dex?.volumeH1))||(marketNum(b.dex?.liquidityUsd)-marketNum(a.dex?.liquidityUsd))).slice(0,180);
    let jupiterOk=false;
    for(const group of chunk(base.map(t=>t.mint),90)){
      if(!group.length)continue;
      try{
        const rows=await fetchJupiterRows(group),byMint=new Map(rows.map(r=>[str(r.id||r.address||r.mint),r]));
        for(const mint of group){const token=tokens.get(mint),row=byMint.get(mint);if(!token||!row)continue;token.jupiter=normalizeJupiterRow(row);scoreToken(token)}
        jupiterOk=true;
      }catch(error){sourceState.vetted.lastError=`Jupiter: ${compactError(error)}`}
      await sleep(350);
    }
    vettedLastRefresh=now();
    if(jupiterOk||base.length===0){touchSource('vetted',true,started);sourceState.vetted.lastError=base.length?'':'No eligible 30d+ DEX markets in current accumulated universe yet'}
    else touchSource('vetted',false,started,sourceState.vetted.lastError||'Jupiter trust pass unavailable');
    const deep=base.filter(t=>{const j=t.jupiter;return j?.isVerified&&marketNum(j.organicScore)>=50&&marketNum(j.liquidity)>=50000&&!j.audit?.isSus&&marketNum(j.audit?.topHoldersPercentage)<65}).slice(0,8);
    for(const token of deep){
      await enrichOnchain(token).catch(()=>{});await sleep(500);
      await enrichSecurity(token).catch(()=>{});await sleep(650);
    }
    vettedLastRefresh=now();
  }catch(error){touchSource('vetted',false,started,error)}
};
