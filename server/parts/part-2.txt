function scoreToken(token) {
  const d = token.dex || {};
  const ageMinutes = ageMs(token) / 60000;
  const liq = num(d.liquidityUsd);
  const mc = num(d.marketCap || token.pump?.marketCapUsd);
  const volH1 = num(d.volumeH1);
  const txM5 = num(d.txM5);
  const buysM5 = num(d.buysM5);
  const sellsM5 = num(d.sellsM5);
  const changeM5 = num(d.changeM5);
  const boosts = num(d.boosts);
  const reasons = [];
  const flags = [];

  let market = 0;
  if (d.pairAddress) market += 3;
  market += clamp(Math.log10(Math.max(liq, 1) / 1000) * 6, 0, 12);
  if (liq >= 10000) reasons.push('liquidity_10k');
  if (liq >= 50000) reasons.push('liquidity_50k');
  const depthRatio = mc > 0 ? liq / mc : 0;
  if (depthRatio >= 0.08 && depthRatio <= 0.8) market += 5;
  else if (mc > 0 && depthRatio < 0.02) flags.push('thin_liquidity');
  if (ageMinutes < 120) market += 2;
  market = clamp(market, 0, 20);

  let momentum = 0;
  momentum += clamp(Math.log10(Math.max(volH1, 1) / 1000) * 4.5, 0, 9);
  momentum += clamp(txM5 / 18, 0, 5);
  if (buysM5 + sellsM5 >= 8) {
    const buyShare = buysM5 / Math.max(1, buysM5 + sellsM5);
    if (buyShare >= 0.54 && buyShare <= 0.82) momentum += 3;
    if (buyShare > 0.93) flags.push('one_sided_flow');
  }
  if (changeM5 > 5 && changeM5 < 80) momentum += 2;
  if (changeM5 > 150) flags.push('vertical_price_move');
  momentum = clamp(momentum, 0, 20);

  let security = 8;
  if (token.chain) {
    security = 8;
    if (!token.chain.mintAuthority) { security += 5; reasons.push('mint_revoked'); }
    else flags.push('mint_authority_active');
    if (!token.chain.freezeAuthority) { security += 4; reasons.push('freeze_revoked'); }
    else flags.push('freeze_authority_active');
    if (String(token.chain.tokenProgram || '').includes('Tokenz')) flags.push('token_2022_review');
  }
  if (token.security?.available) {
    if (token.security.creatorMalicious) flags.push('creator_flagged');
    if (token.security.riskItems?.length) flags.push(...token.security.riskItems.map(x => `goplus_${x}`));
    if (!token.security.riskItems?.length && !token.security.creatorMalicious) security += 3;
  }
  security = clamp(security, 0, 20);

  let distribution = 10;
  const h = token.holders;
  if (h && h.stage !== 'bonding-curve') {
    distribution = 20;
    if (h.top10Pct > 70) { distribution -= 12; flags.push('top10_extreme'); }
    else if (h.top10Pct > 50) { distribution -= 7; flags.push('top10_concentrated'); }
    else if (h.top10Pct > 35) distribution -= 3;
    else reasons.push('distribution_broad');
    if (h.top1Pct > 25) { distribution -= 5; flags.push('top1_large'); }
  }
  distribution = clamp(distribution, 0, 20);

  let transparency = 0;
  if (token.links.website) transparency += 5;
  if (token.links.twitter) transparency += 5;
  if (token.links.telegram) transparency += 2;
  if (token.description && token.description.length > 30) transparency += 2;
  if (token.creator) transparency += 1;
  if (token.creatorStats?.serialDeployer) flags.push('serial_deployer');
  if (token.creatorStats?.projectCount > 0 && token.creatorStats?.projectCount <= 4) transparency += 1;
  transparency = clamp(transparency, 0, 15);

  let attention = clamp(boosts / 4, 0, 4);
  if (boosts >= 10) reasons.push('dex_boosts');
  if (token.cex?.binanceMention) { attention += 5; reasons.push('binance_public_mention'); }
  if (token.pump?.complete) { attention += 3; reasons.push('pump_graduated'); }
  attention = clamp(attention, 0, 5);

  let score = Math.round(market + momentum + security + distribution + transparency + attention);
  const severe = flags.filter(f => ['creator_flagged','mint_authority_active','freeze_authority_active','top10_extreme'].includes(f)).length;
  if (severe) score -= severe * 7;
  if (!d.pairAddress && ageMinutes > 15) score = Math.min(score, 42);
  if (liq > 0 && liq < 1500 && ageMinutes > 10) score = Math.min(score, 45);
  score = clamp(score, 0, 100);

  let risk = 45;
  if (!token.chain) risk += 5;
  if (flags.includes('mint_authority_active')) risk += 18;
  if (flags.includes('freeze_authority_active')) risk += 16;
  if (flags.includes('creator_flagged')) risk += 30;
  if (flags.includes('top10_extreme')) risk += 20;
  if (flags.includes('top10_concentrated')) risk += 10;
  if (flags.includes('thin_liquidity')) risk += 8;
  if (flags.includes('serial_deployer')) risk += 8;
  if (reasons.includes('mint_revoked')) risk -= 8;
  if (reasons.includes('freeze_revoked')) risk -= 7;
  if (reasons.includes('distribution_broad')) risk -= 7;
  risk = clamp(Math.round(risk), 0, 100);

  let riskLevel = 'unknown';
  if (token.chain || token.security || token.holders) {
    riskLevel = risk >= 75 ? 'critical' : risk >= 58 ? 'high' : risk >= 38 ? 'medium' : 'low';
  }

  const previous = token.score || 0;
  token.score = score;
  token.scoreParts = {
    market: Math.round(market),
    momentum: Math.round(momentum),
    security: Math.round(security),
    distribution: Math.round(distribution),
    transparency: Math.round(transparency),
    attention: Math.round(attention)
  };
  token.risk = risk;
  token.riskLevel = riskLevel;
  token.reasons = uniq(reasons).slice(0, 10);
  token.flags = uniq(flags).slice(0, 12);
  token.tier = score >= 85 ? 'prime' : score >= 70 ? 'candidate' : score >= 55 ? 'watch' : 'raw';

  const eligible = liq >= 7500 && d.pairAddress && !['critical'].includes(riskLevel);
  if (eligible && previous < 70 && score >= 70 && !token.alertedCandidate) {
    token.alertedCandidate = true;
    addAlert({
      severity: score >= 85 ? 'critical' : 'important',
      type: 'candidate',
      mint: token.mint,
      symbol: token.symbol,
      title: 'RADAR_CANDIDATE',
      message: `Radar score crossed ${score}`,
      score
    });
  }
  if (eligible && previous < 85 && score >= 85 && !token.alertedPrime) {
    token.alertedPrime = true;
    addAlert({
      severity: 'critical',
      type: 'prime',
      mint: token.mint,
      symbol: token.symbol,
      title: 'PRIME_WATCH',
      message: `High-signal configuration detected (${score})`,
      score
    });
  }
}

async function initDb() {
  if (!pool) return;
  await pool.query(`CREATE TABLE IF NOT EXISTS radar_tokens (
    mint text PRIMARY KEY,
    payload jsonb NOT NULL,
    score integer NOT NULL DEFAULT 0,
    updated_at bigint NOT NULL
  )`);
  await pool.query(`CREATE TABLE IF NOT EXISTS radar_alerts (
    id text PRIMARY KEY,
    payload jsonb NOT NULL,
    created_at bigint NOT NULL
  )`);
  const { rows } = await pool.query('SELECT payload FROM radar_tokens ORDER BY updated_at DESC LIMIT 400');
  for (const row of rows) {
    const t = row.payload;
    if (t?.mint) tokens.set(t.mint, { ...createToken(t.mint), ...t, source: t.source || [] });
  }
}

async function persistToken(token) {
  if (!pool) return;
  const payload = {
    ...token,
    timeline: token.timeline.slice(-64),
    security: token.security ? { ...token.security, raw: undefined } : null
  };
  await pool.query(
    'INSERT INTO radar_tokens (mint,payload,score,updated_at) VALUES ($1,$2,$3,$4) ON CONFLICT (mint) DO UPDATE SET payload=EXCLUDED.payload,score=EXCLUDED.score,updated_at=EXCLUDED.updated_at',
    [token.mint, payload, token.score, now()]
  );
}

async function persistAlert(alert) {
  if (!pool) return;
  await pool.query('INSERT INTO radar_alerts (id,payload,created_at) VALUES ($1,$2,$3) ON CONFLICT DO NOTHING', [alert.id, alert, alert.at]);
}

function upsertPump(coin) {
  const mint = str(coin?.mint || coin?.address || coin?.token_address);
  if (!mint || mint.length < 30) return null;
  let token = tokens.get(mint);
  const isNew = !token;
  if (!token) token = createToken(mint);
  token.name = str(coin?.name) || token.name;
  token.symbol = str(coin?.symbol) || token.symbol;
  token.description = str(coin?.description) || token.description;
  token.image = safeUrl(coin?.image_uri || coin?.image) || token.image;
  token.creator = str(coin?.creator) || token.creator;
  token.createdAt = normalizeTs(coin?.created_timestamp || coin?.createdAt) || token.createdAt;
  token.lastUpdatedAt = now();
  token.source = uniq([...(token.source || []), 'pump.fun']);
  token.pump = {
    ...(token.pump || {}),
    complete: Boolean(coin?.complete),
    replyCount: num(coin?.reply_count || coin?.replyCount),
    marketCapUsd: num(coin?.usd_market_cap || coin?.market_cap_usd),
    marketCapSol: num(coin?.market_cap),
    bondingCurve: str(coin?.bonding_curve || coin?.associated_bonding_curve),
    lastTradeAt: normalizeTs(coin?.last_trade_timestamp),
    kingAt: normalizeTs(coin?.king_of_the_hill_timestamp),
    nsfw: Boolean(coin?.nsfw)
  };
  mergeSocials(token, coin);
  tokens.set(mint, token);
  scoreToken(token);
  if (isNew) broadcast('new-token', publicToken(token));
  return token;
}

function upsertDexProfile(profile) {
  if (str(profile?.chainId).toLowerCase() !== 'solana') return null;
  const mint = str(profile?.tokenAddress);
  if (!mint) return null;
  let token = tokens.get(mint);
  const isNew = !token;
  if (!token) token = createToken(mint);
  token.source = uniq([...(token.source || []), 'dexscreener']);
  token.description ||= str(profile?.description);
  token.image ||= safeUrl(profile?.icon || profile?.header);
  token.lastUpdatedAt = now();
  const socials = [];
  const websites = [];
  for (const link of profile?.links || []) {
    const type = str(link?.type || link?.label).toLowerCase();
    const url = safeUrl(link?.url);
    if (!url) continue;
