    if (type.includes('twitter') || type === 'x') socials.push({ platform: 'twitter', url });
    else if (type.includes('telegram')) socials.push({ platform: 'telegram', url });
    else websites.push({ url });
  }
  mergeSocials(token, { dexSocials: socials, dexWebsites: websites });
  tokens.set(mint, token);
  if (isNew) broadcast('new-token', publicToken(token));
  return token;
}

function applyDexPair(token, pair) {
  if (!token || !pair) return;
  const previousLiq = token.dex?.liquidityUsd || 0;
  const previousVol = token.dex?.volumeH1 || 0;
  const txM5 = pair?.txns?.m5 || {};
  const txH1 = pair?.txns?.h1 || {};
  token.name ||= str(pair?.baseToken?.name);
  token.symbol ||= str(pair?.baseToken?.symbol);
  token.lastUpdatedAt = now();
  token.source = uniq([...(token.source || []), 'dexscreener']);
  token.dex = {
    dexId: str(pair?.dexId),
    pairAddress: str(pair?.pairAddress),
    url: safeUrl(pair?.url),
    priceUsd: num(pair?.priceUsd),
    marketCap: num(pair?.marketCap),
    fdv: num(pair?.fdv),
    liquidityUsd: num(pair?.liquidity?.usd),
    volumeM5: num(pair?.volume?.m5),
    volumeH1: num(pair?.volume?.h1),
    volumeH6: num(pair?.volume?.h6),
    txM5: num(txM5.buys) + num(txM5.sells),
    buysM5: num(txM5.buys),
    sellsM5: num(txM5.sells),
    txH1: num(txH1.buys) + num(txH1.sells),
    buysH1: num(txH1.buys),
    sellsH1: num(txH1.sells),
    changeM5: num(pair?.priceChange?.m5),
    changeH1: num(pair?.priceChange?.h1),
    boosts: num(pair?.boosts?.active),
    pairCreatedAt: normalizeTs(pair?.pairCreatedAt)
  };
  token.image ||= safeUrl(pair?.info?.imageUrl);
  mergeSocials(token, { dexSocials: pair?.info?.socials || [], dexWebsites: pair?.info?.websites || [] });
  scoreToken(token);
  recordSnapshot(token);

  const liq = token.dex.liquidityUsd;
  const vol = token.dex.volumeH1;
  if (previousLiq > 3000 && liq > previousLiq * 1.5 && liq - previousLiq > 5000) {
    addAlert({ severity: 'important', type: 'liquidity', mint: token.mint, symbol: token.symbol, title: 'LIQUIDITY_SURGE', message: `Liquidity $${Math.round(previousLiq).toLocaleString()} → $${Math.round(liq).toLocaleString()}`, score: token.score });
  }
  if (previousVol > 5000 && vol > previousVol * 1.8 && vol - previousVol > 10000) {
    addAlert({ severity: 'important', type: 'volume', mint: token.mint, symbol: token.symbol, title: 'VOLUME_ACCELERATION', message: `1h volume accelerated`, score: token.score });
  }
}

async function pollPump() {
  const started = now();
  try {
    const url = 'https://frontend-api-v3.pump.fun/coins?offset=0&limit=50&sort=created_timestamp&order=DESC&includeNsfw=false';
    const data = await fetchJson(url, { timeout: 8000 });
    const list = extractPumpArray(data);
    if (!list.length) throw new Error('Empty new-token feed');
    for (const coin of list) upsertPump(coin);
    touchSource('pump', true, started);
    pruneTokens();
  } catch (error) {
    touchSource('pump', false, started, error);
  }
}

async function pollDexProfiles() {
  const started = now();
  try {
    const data = await fetchJson('https://api.dexscreener.com/token-profiles/latest/v1');
    const list = Array.isArray(data) ? data : data ? [data] : [];
    for (const profile of list) upsertDexProfile(profile);
    touchSource('dex', true, started);
  } catch (error) {
    touchSource('dex', false, started, error);
  }
}

async function pollDexPairs() {
  const started = now();
  try {
    const candidates = [...tokens.values()]
      .filter(t => ageMs(t) < 24 * 3600_000)
      .sort((a, b) => (b.score - a.score) || (b.discoveredAt - a.discoveredAt))
      .slice(0, 180);
    const mints = candidates.map(t => t.mint);
    for (const group of chunk(mints, 30).slice(0, 6)) {
      if (!group.length) continue;
      const data = await fetchJson(`https://api.dexscreener.com/tokens/v1/solana/${group.join(',')}`, { timeout: 8000 });
      const pairs = Array.isArray(data) ? data : [];
      const best = new Map();
      for (const pair of pairs) {
        const mint = str(pair?.baseToken?.address);
        if (!mint || !tokens.has(mint)) continue;
        const liq = num(pair?.liquidity?.usd);
        if (!best.has(mint) || liq > num(best.get(mint)?.liquidity?.usd)) best.set(mint, pair);
      }
      for (const [mint, pair] of best) applyDexPair(tokens.get(mint), pair);
      await sleep(150);
    }
    touchSource('dex', true, started);
  } catch (error) {
    touchSource('dex', false, started, error);
  }
}

function parseGoPlus(raw, mint) {
  const result = raw?.result || {};
  const info = result[mint] || result?.[mint.toLowerCase?.()] || (Array.isArray(result) ? result[0] : null);
  if (!info || typeof info !== 'object') return { available: false, riskItems: [], checkedAt: now() };
  const riskItems = [];
  const statusOne = value => value === '1' || value === 1 || value === true || value?.status === '1' || value?.status === 1;
  if (statusOne(info.mintable)) riskItems.push('mintable');
  if (statusOne(info.freezable)) riskItems.push('freezable');
  if (statusOne(info.non_transferable)) riskItems.push('non_transferable');
  if (statusOne(info.closable)) riskItems.push('closable');
  if (statusOne(info.balance_mutable_authority)) riskItems.push('balance_mutable');
  if (statusOne(info.transfer_fee_upgradable)) riskItems.push('fee_upgradable');
  if (statusOne(info.default_account_state_upgradable)) riskItems.push('account_state_upgradable');
  const creatorRows = Array.isArray(info.creator) ? info.creator : info.creator ? [info.creator] : [];
  const creatorMalicious = creatorRows.some(x => statusOne(x?.malicious_address));
  const holders = Array.isArray(info.holders) ? info.holders : [];
  const top10Pct = holders.slice(0, 10).reduce((sum, h) => sum + num(h?.percent) * (num(h?.percent) <= 1 ? 100 : 1), 0);
  return {
    available: true,
    riskItems: uniq(riskItems),
    holderCount: num(info.holder_count),
    top10Pct: Math.round(top10Pct * 100) / 100,
    creatorMalicious,
    checkedAt: now(),
    raw: {
      trusted_token: info.trusted_token,
      mintable: info.mintable,
      freezable: info.freezable,
      non_transferable: info.non_transferable,
      closable: info.closable,
      metadata_mutable: info.metadata_mutable,
      transfer_fee: info.transfer_fee,
      balance_mutable_authority: info.balance_mutable_authority,
      holder_count: info.holder_count
    }
  };
}

async function enrichSecurity(token) {
  if (!token || now() - token.lastSecurityAt < 20 * 60_000) return;
  token.lastSecurityAt = now();
  const started = now();
  try {
    const headers = GOPLUS_TOKEN ? { authorization: `Bearer ${GOPLUS_TOKEN}` } : {};
    const data = await fetchJson(`https://api.gopluslabs.io/api/v1/solana/token_security?contract_addresses=${encodeURIComponent(token.mint)}`, { headers });
    token.security = parseGoPlus(data, token.mint);
    touchSource('goplus', true, started);
    scoreToken(token);
  } catch (error) {
    token.security ||= { available: false, riskItems: [], checkedAt: now() };
    touchSource('goplus', false, started, error);
  }
}

async function enrichOnchain(token) {
  if (!token || now() - token.lastEnrichedAt < 15 * 60_000) return;
  token.lastEnrichedAt = now();
  try {
    const mintInfo = await rpc('getAccountInfo', [token.mint, { encoding: 'jsonParsed', commitment: 'confirmed' }]);
    const parsed = mintInfo?.value?.data?.parsed?.info || {};
    const supply = num(parsed?.supply);
    const decimals = num(parsed?.decimals);
    token.chain = {
      mintAuthority: parsed?.mintAuthority || null,
      freezeAuthority: parsed?.freezeAuthority || null,
      decimals,
      supply,
      tokenProgram: str(mintInfo?.value?.owner)
    };

    const largest = await rpc('getTokenLargestAccounts', [token.mint, { commitment: 'confirmed' }]);
    const rows = Array.isArray(largest?.value) ? largest.value.slice(0, 20) : [];
    const addresses = rows.map(x => x.address).filter(Boolean);
    let owners = [];
    if (addresses.length) {
      const multiple = await rpc('getMultipleAccounts', [addresses, { encoding: 'jsonParsed', commitment: 'confirmed' }]);
      owners = (multiple?.value || []).map((account, index) => ({
        tokenAccount: addresses[index],
        owner: account?.data?.parsed?.info?.owner || '',
        amount: num(account?.data?.parsed?.info?.tokenAmount?.amount || rows[index]?.amount),
        uiAmount: num(account?.data?.parsed?.info?.tokenAmount?.uiAmountString || rows[index]?.uiAmountString)
      }));
    }
    const supplyUi = supply / Math.pow(10, decimals || 0);
    const top = owners.map(x => ({
      ...x,
      pct: supplyUi > 0 ? (x.uiAmount / supplyUi) * 100 : 0
    })).sort((a, b) => b.pct - a.pct);
    const stage = token.pump && !token.pump.complete ? 'bonding-curve' : 'market';
    const topPct = n => top.slice(0, n).reduce((sum, x) => sum + x.pct, 0);
    token.holders = {
      top1Pct: Math.round(topPct(1) * 100) / 100,
      top5Pct: Math.round(topPct(5) * 100) / 100,
      top10Pct: Math.round(topPct(10) * 100) / 100,
      holderCount: token.security?.holderCount || 0,
      stage,
      top: top.slice(0, 10),
      note: stage === 'bonding-curve' ? 'Concentration is not scored while Pump.fun bonding curve still holds inventory.' : 'Top-holder concentration from public Solana RPC.'
    };
    scoreToken(token);
    recordSnapshot(token);
  } catch (error) {
    token.lastEnrichedAt = now() - 12 * 60_000;
  }
}
