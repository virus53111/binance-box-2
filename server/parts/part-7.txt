// Reads token-provided public channels through Jina Reader so arbitrary project URLs are never fetched directly by this server.
sourceState.project=sourceState.project||{label:'Project Channels',ok:false,lastOk:0,lastError:'',latency:null};
const projectSignalState=new Map();
const PROJECT_SCAN_INTERVAL=8*60_000;
const PROJECT_MAX_TOKENS_PER_PASS=10;
const PROJECT_READER_GAP=3200;
let lastProjectReaderAt=0;
const projectExchangeNames=['Binance','Bybit','OKX','Coinbase','Kraken','KuCoin','Gate.io','MEXC','Bitget','HTX','Crypto.com','Upbit','Bithumb'];
const projectFutureListingRe=/(will\s+(?:be\s+)?list(?:ed|ing)|to\s+be\s+list(?:ed|ing)|listing\s+(?:on|at|soon)|cex\s+listing|trading\s+(?:will\s+)?(?:start|open|begin)|deposits?\s+(?:will\s+)?open|coming\s+(?:soon\s+)?to|launch(?:ing|es)?\s+on|debut(?:ing|s)?\s+on|gets?\s+listed|new\s+exchange\s+listing)/i;
const projectExchangeRe=new RegExp(`\\b(${projectExchangeNames.map(escapeRegExp).join('|')})\\b`,'i');
function projectSourceList(token){const out=[];const add=(type,url)=>{url=safeUrl(url);if(!url||!/^https:\/\//i.test(url))return;out.push({type,url})};add('website',token.links?.website);add('x',token.links?.twitter);add('telegram',token.links?.telegram);return out.slice(0,3)}
function readerUrl(url){return `https://r.jina.ai/${url}`}
async function fetchProjectChannel(source){const gap=PROJECT_READER_GAP-(now()-lastProjectReaderAt);if(gap>0)await sleep(gap);lastProjectReaderAt=now();return await fetchText(readerUrl(source.url),{timeout:16000,headers:{'x-timeout':'12','x-token-budget':'5000'}})}
function extractProjectListingSignals(text,source){const raw=String(text||'').replace(/\r/g,'\n'),segments=raw.split(/\n+|(?<=[.!?])\s+(?=[A-Z0-9@#])/).map(x=>x.replace(/\s+/g,' ').trim()).filter(x=>x.length>=18&&x.length<=650),hits=[];for(const segment of segments){if(!projectFutureListingRe.test(segment))continue;const exchange=segment.match(projectExchangeRe)?.[1]||'';if(!exchange)continue;const stronger=/will\s+(?:be\s+)?listed|to\s+be\s+listed|trading\s+(?:will\s+)?(?:start|open|begin)|listing\s+on/i.test(segment);hits.push({exchange,kind:'project_listing_claim',evidenceScore:source.type==='x'||source.type==='telegram'?(stronger?88:80):(stronger?82:74),contractVerified:false,projectClaim:true,sourceType:source.type,title:`Project channel mentions a future ${exchange} listing/trading event`,excerpt:segment.slice(0,420),url:source.url,at:now(),note:'Token-provided public channel. This is a project claim, not confirmation by the exchange.'})}const dedup=new Map();for(const h of hits){const key=`${h.exchange}:${h.excerpt.toLowerCase().slice(0,180)}`;if(!dedup.has(key))dedup.set(key,h)}return[...dedup.values()].slice(0,6)}
function projectFingerprint(signal){let h=2166136261;const s=`${signal.exchange}|${signal.excerpt}`.toLowerCase();for(let i=0;i<s.length;i++){h^=s.charCodeAt(i);h=Math.imul(h,16777619)}return(h>>>0).toString(16)}
async function scanProjectToken(token){const sources=projectSourceList(token);if(!sources.length)return;const previous=projectSignalState.get(token.mint)||{fingerprints:new Set(),signals:[],initialized:false,lastScan:0,errors:[]},newSignals=[],errors=[];let ok=0;for(const source of sources){try{const text=await fetchProjectChannel(source);if(!text||text.length<80)throw new Error('Reader returned too little text');ok++;for(const s of extractProjectListingSignals(text,source))newSignals.push(s)}catch(error){errors.push(`${source.type}: ${compactError(error)}`)}}const merged=new Map();for(const s of [...newSignals,...previous.signals])merged.set(projectFingerprint(s),s);const signals=[...merged.values()].sort((a,b)=>b.evidenceScore-a.evidenceScore).slice(0,12),currentFps=new Set(newSignals.map(projectFingerprint));if(previous.initialized){for(const signal of newSignals){const fp=projectFingerprint(signal);if(previous.fingerprints.has(fp))continue;addAlert({severity:signal.evidenceScore>=86?'critical':'important',type:'project-listing',mint:token.mint,symbol:token.symbol,title:'PROJECT_LISTING_CLAIM',message:`${signal.exchange}: ${signal.excerpt.slice(0,180)}`,score:token.score,sourceUrl:signal.url})}}projectSignalState.set(token.mint,{fingerprints:currentFps,signals,initialized:true,lastScan:now(),errors,okSources:ok,totalSources:sources.length});if(ok)touchSource('project',true,now()-1);else if(errors.length)touchSource('project',false,now()-1,errors.join('; '))}
async function projectChannelLoop(){while(true){const started=now();try{const queue=[...tokens.values()].filter(t=>projectSourceList(t).length&&ageMs(t)<30*24*3600_000&&(t.score>=52||t.dex?.liquidityUsd>=10000)).sort((a,b)=>((b.score||0)-(a.score||0))||((b.dex?.liquidityUsd||0)-(a.dex?.liquidityUsd||0)));for(const token of queue.slice(0,PROJECT_MAX_TOKENS_PER_PASS)){const st=projectSignalState.get(token.mint);if(st&&now()-st.lastScan<PROJECT_SCAN_INTERVAL)continue;await scanProjectToken(token)}if(queue.length===0)sourceState.project.lastError='Waiting for eligible tokens with public project channels'}catch(error){touchSource('project',false,started,error)}await sleep(PROJECT_SCAN_INTERVAL)}}
const baseGetCexIntelProject=getCexIntel;getCexIntel=function(token){const base=baseGetCexIntelProject(token),project=projectSignalState.get(token.mint),claims=project?.signals||[],combined=[...claims,...(base.signals||[])].sort((a,b)=>(b.evidenceScore||0)-(a.evidenceScore||0)).slice(0,12),bestProject=claims[0]||null,best=Math.max(base.evidenceScore||0,bestProject?.evidenceScore||0);return{...base,signals:combined,projectSignals:claims,projectChannels:{lastScan:project?.lastScan||0,okSources:project?.okSources||0,totalSources:project?.totalSources||0,errors:project?.errors||[]},preCexCandidate:base.preCexCandidate||Boolean(!base.majorCexSymbolDetected&&bestProject&&bestProject.evidenceScore>=74),evidenceScore:best,evidenceLevel:bestProject&&bestProject.evidenceScore>=86?'project_strong_claim':base.evidenceLevel,projectClaimDetected:Boolean(bestProject)}};

app.get('/api/project-signals/:mint',(req,res)=>{const mint=str(req.params.mint),token=tokens.get(mint);if(!token)return res.status(404).json({error:'Token not found yet'});const project=projectSignalState.get(mint)||null;res.json({generatedAt:now(),mint,project,cex:getCexIntel(token)})});
setTimeout(()=>projectChannelLoop().catch(()=>{}),12000);
