function touchSource(key, ok, started, error = '') {
  const s = sourceState[key];
  if (!s) return;
  s.ok = ok;
  s.latency = Math.max(0, now() - started);
  if (ok) {
    s.lastOk = now();
    s.lastError = '';
  } else {
    s.lastError = compactError(error);
  }
}

async function fetchJson(url, options = {}) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), options.timeout || FETCH_TIMEOUT);
  try {
    const response = await fetch(url, {
      ...options,
      headers: {
        accept: 'application/json,text/plain,*/*',
        'user-agent': 'Mozilla/5.0 (compatible; MurdiliRadar/1.0; +https://murdilimax.com)',
        ...(options.headers || {})
      },
      signal: controller.signal
    });
    if (!response.ok) throw new Error(`${response.status} ${response.statusText}`);
    const text = await response.text();
    return text ? JSON.parse(text) : null;
  } finally {
    clearTimeout(timer);
  }
}

async function rpc(method, params) {
  const started = now();
  try {
    const data = await fetchJson(SOLANA_RPC_URL, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ jsonrpc: '2.0', id: 1, method, params })
    });
    if (data?.error) throw new Error(data.error.message || 'RPC error');
    touchSource('solana', true, started);
    return data?.result;
  } catch (error) {
    touchSource('solana', false, started, error);
    throw error;
  }
}

function safeUrl(value) {
  const raw = str(value);
  if (!raw) return '';
  try {
    const url = new URL(raw.startsWith('http') ? raw : `https://${raw}`);
    return ['http:', 'https:'].includes(url.protocol) ? url.toString() : '';
  } catch {
    return '';
  }
}

function extractPumpArray(payload) {
  if (Array.isArray(payload)) return payload;
  if (Array.isArray(payload?.data)) return payload.data;
  if (Array.isArray(payload?.coins)) return payload.coins;
  return [];
}

function mergeSocials(token, data = {}) {
  const links = {
    website: safeUrl(data.website || token.links?.website),
    twitter: safeUrl(data.twitter || token.links?.twitter),
    telegram: safeUrl(data.telegram || token.links?.telegram)
  };
  for (const item of data.dexSocials || []) {
    const platform = str(item?.platform).toLowerCase();
    const url = safeUrl(item?.url || item?.handle);
    if (!url) continue;
    if (platform.includes('twitter') || platform === 'x') links.twitter ||= url;
    if (platform.includes('telegram')) links.telegram ||= url;
  }
  for (const item of data.dexWebsites || []) links.website ||= safeUrl(item?.url);
  token.links = links;
}

function broadcast(type, payload) {
  const data = `event: ${type}\ndata: ${JSON.stringify(payload)}\n\n`;
  for (const res of clients) res.write(data);
}

function addAlert(alert) {
  const item = {
    id: `${now()}-${Math.random().toString(36).slice(2, 8)}`,
    at: now(),
    severity: 'info',
    ...alert
  };
  alerts.unshift(item);
  if (alerts.length > MAX_ALERTS) alerts.length = MAX_ALERTS;
  broadcast('alert', item);
  persistAlert(item).catch(() => {});
  return item;
}

function recordSnapshot(token) {
  const point = {
    at: now(),
    mc: token.dex?.marketCap || token.pump?.marketCapUsd || 0,
    liq: token.dex?.liquidityUsd || 0,
    vol: token.dex?.volumeH1 || 0,
    score: token.score || 0
  };
  const prev = token.timeline[token.timeline.length - 1];
  if (!prev || point.at - prev.at > 25000 || Math.abs(point.mc - prev.mc) / Math.max(prev.mc, 1) > 0.08) {
    token.timeline.push(point);
    if (token.timeline.length > 96) token.timeline.shift();
  }
}
